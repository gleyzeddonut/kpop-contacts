# Briefs Feature Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add PDF brief import — drop a label's brief PDF into the app, Claude extracts structured data, briefs are matched to artists and displayed in a new "Briefs" tab on the workspace.

**Architecture:** New `briefs` Supabase table (one row per artist per PDF). Claude API (already integrated in `main.js`) reads the PDF via base64 document block and returns structured JSON. The renderer matches artist names fuzzy-style, auto-creates unmatched artists, and renders briefs in a new tab inside the existing `renderDetail()` workspace.

**Tech Stack:** Electron 35, Supabase JS, Claude API (`claude-sonnet-4-6`), vanilla JS/CSS in `index.html`.

```sql
create table public.briefs (

  list_id              uuid references public.lists(id) on delete cascade not null,
  artist_id            uuid references public.artists(id) on delete cascade not null,

alter table public.briefs enable row level security;

-- Owner has full access
create policy "list owner full access"
  on public.briefs for all
  using (
    exists (select 1 from public.lists where id = briefs.list_id and owner_id = auth.uid())
  )
  with check (
    exists (select 1 from public.lists where id = briefs.list_id and owner_id = auth.uid())
  );

-- Editors can read and write
create policy "editors can read and write briefs"
  on public.briefs for all
  using (
    exists (
      select 1 from public.list_shares
      where list_id = briefs.list_id

        and role = 'editor'
    )
  )

      where list_id = briefs.list_id
        and shared_with_email = (auth.jwt() ->> 'email')
```

```js
  const { list_id, created_at, ...rest } = brief

  )
  if (error) throw error
}

async function deleteBrief(briefId) {
  const { error } = await client().from('briefs').delete().eq('id', briefId)
  if (error) throw error
}
```

```js
// ── Claude API: parse a brief PDF ──
async function parseBriefPdf(filePath) {
  const buffer = fs.readFileSync(filePath)
  const base64 = buffer.toString('base64')

  const prompt = `Extract all brief information from this PDF and return valid JSON matching this schema exactly:
{
  "label": "string",
  "submission_emails": ["string"],
  "artists": [
    {
      "name": "string",
      "deadline": "string or null",
      "general_direction": "string or null",
      "track_types": [
        {
          "name": "string",
          "tags": ["string"],
          "wants": ["string"],
          "avoids": ["string"],
          "references": [{ "title": "string", "url": "string" }]
        }
      ]
    }
  ]
}

Rules:
- label: the issuing label name (e.g. "SM Entertainment").
- submission_emails: all email addresses from the cover/intro page.
- artists: one entry per artist section in the PDF.
- deadline: extract as-is ("ASAP", "By May 8th", "early June"). null if not stated.
- general_direction: the [General Direction] block if present, else null.
- tags: hashtags found in the track type heading (e.g. #GenZ_Energy). Empty array if none.
- wants: bullet points describing what they want. Also include items under "Please include:".
- avoids: bullet points under "Please avoid:" or explicit negations in direction text.
- references: YouTube/music URLs with their track title. Empty array if "No specific references provided".
Return ONLY the JSON object. No markdown, no code fences, no explanation.`

  const data = await callClaude({
    model: 'claude-sonnet-4-6',
    max_tokens: 4096,
    messages: [{
      role: 'user',
      content: [
        {
          type: 'document',
          source: {
            type: 'base64',
            media_type: 'application/pdf',
            data: base64,
          },
        },
        { type: 'text', text: prompt },
      ],
    }],
  })

  const text = (data.content || [])
    .filter(b => b.type === 'text')
    .map(b => b.text)
    .join('')
  if (!text) throw new Error('No response from Claude')

  const match = text.match(/\{[\s\S]*\}/)
  if (!match) throw new Error('Claude did not return a JSON object')

  try { return JSON.parse(match[0]) }
  catch { throw new Error('Claude returned unparseable JSON') }
}
```

```js
function normalizeBriefArtistName(name) {
  // Strip trailing parentheticals: "MINHO (of SHINee)" → "MINHO"
  return name.replace(/\s*\(.*?\)\s*$/, '').trim().toLowerCase();
}
```

- [ ] **Step 4: Add `importBriefPdf` function**

```js
async function importBriefPdf() {
  if (!currentListId) { toast('Open a list first', true); return; }
  toast('Importing brief PDF…');
  let parsed;
  try {
    parsed = await window.kc.importBriefPdf();
  } catch (err) {
    toast('Import failed: ' + err.message, true);
    return;
  }
  if (!parsed) return; // user cancelled dialog

  const { label, submission_emails: submissionEmails, artists: briefArtists, _sourcePdf } = parsed;
  if (!Array.isArray(briefArtists) || briefArtists.length === 0) {
    toast('No artist briefs found in this PDF', true);
    return;
  }

  // Match submission emails to existing contacts
  const safeEmails = Array.isArray(submissionEmails) ? submissionEmails : [];
  const matchedContactIds = contacts
    .filter(c => c.email && safeEmails.some(e => e.toLowerCase() === c.email.toLowerCase()))
    .map(c => c.id);

  let importedCount = 0;
  const errors = [];

  for (const ba of briefArtists) {
    try {
      let artist = matchArtistByName(ba.name);
      if (!artist) {
        artist = await autoCreateArtistFromBrief(ba.name, label);
      }

      const brief = {
        id: uid(),
        list_id: currentListId,
        artist_id: artist.id,
        source_pdf: _sourcePdf || '',
        label: label || '',
        deadline: ba.deadline || null,
        general_direction: ba.general_direction || null,
        track_types: Array.isArray(ba.track_types) ? ba.track_types : [],
        submission_emails: safeEmails,
        matched_contact_ids: matchedContactIds,
      };
      briefs.push(brief);
      await syncBrief(brief);
      importedCount++;
    } catch (err) {
      errors.push(ba.name + ': ' + err.message);
    }
  }

  render();
  if (errors.length > 0) {
    toast(`Imported ${importedCount} briefs · ${errors.length} failed`, true);
  } else {
    const contactNote = matchedContactIds.length > 0
      ? ` · ${matchedContactIds.length} contact${matchedContactIds.length !== 1 ? 's' : ''} matched`
      : '';
    toast(`Brief imported: ${importedCount} artist${importedCount !== 1 ? 's' : ''}${contactNote}`);
  }
}
```

```js
function deadlineBadgeClass(deadline) {
  if (!deadline) return null;
  const d = deadline.toUpperCase();
  if (d.includes('ASAP')) return 'asap';
  // Try to parse a date
  const parsed = new Date(deadline.replace(/By\s*/i, '').trim());
  if (!isNaN(parsed)) {
    return parsed < new Date() ? 'overdue' : 'future';
  }
  return 'asap'; // fallback for ambiguous strings
}

function renderBriefsTab(artist) {
  const pane = document.getElementById('briefs-pane');
  if (!pane) return;
  pane.className = 'workspace-table-area briefs-pane';

  const artistBriefs = briefs.filter(b => b.artist_id === artist.id);

  if (artistBriefs.length === 0) {
    pane.innerHTML = `<div class="brief-empty">No briefs yet.<br>Click "Import Brief PDF" to add one.</div>`;
    return;
  }

  pane.innerHTML = artistBriefs.map(brief => {
    const badgeClass = deadlineBadgeClass(brief.deadline);
    const deadlineBadge = badgeClass && brief.deadline
      ? `<span class="deadline-badge ${badgeClass}">${h(brief.deadline)}</span>`
      : '';

    const emailPreview = (brief.submission_emails || []).slice(0, 3)
      .map(e => e.split('@')[0]).join(' · ');
    const emailsHTML = emailPreview
      ? `<span class="brief-emails">${h(emailPreview)}</span>`
      : '';

    const directionHTML = brief.general_direction
      ? `<div class="brief-section">
           <div class="brief-section-label">General Direction</div>
           <div class="brief-direction-text">${h(brief.general_direction)}</div>
         </div>`
      : '';

    const trackTypes = Array.isArray(brief.track_types) ? brief.track_types : [];
    const trackTypesHTML = trackTypes.length === 0 ? '' : `
      <div class="brief-section">
        <div class="brief-section-label">Track Types</div>
        ${trackTypes.map(tt => {
          const wants = (tt.wants || []).map(w => `· ${h(w)}`).join('<br>');
          const avoids = (tt.avoids || []).map(a => `✕ ${h(a)}`).join('<br>');
          const refs = (tt.references || [])
            .filter(r => r.url && (r.url.startsWith('https://') || r.url.startsWith('http://')))
            .map(r => `<a class="ref-pill" href="#" onclick="event.preventDefault();window.open('${h(r.url)}')">▶ ${h(r.title || r.url)}</a>`)
            .join('');
          const tagsLine = (tt.tags || []).length > 0
            ? `<div style="font-size:9px;color:var(--text-dim);font-family:var(--font-mono);margin-bottom:4px">${tt.tags.map(t => h(t)).join(' ')}</div>`
            : '';
          return `
            <div class="track-type-card">
              <div class="track-type-name">${h(tt.name || 'Track Type')}</div>
              ${tagsLine}
              ${wants ? `<div class="track-type-wants">${wants}</div>` : ''}
              ${avoids ? `<div class="track-type-avoids">${avoids}</div>` : ''}
              ${refs ? `<div class="track-type-refs">${refs}</div>` : ''}
            </div>`;
        }).join('')}
      </div>`;

    const matchedIds = brief.matched_contact_ids || [];
    const matchedBanner = matchedIds.length > 0
      ? `<div class="brief-matched-contacts">✓ ${matchedIds.length} contact${matchedIds.length !== 1 ? 's' : ''} matched from brief emails</div>`
      : '';

    return `
      <div class="brief-card">
        <div class="brief-card-header">
          <span class="brief-source">${h(brief.source_pdf || brief.label || 'Brief')}</span>
          ${deadlineBadge}
          ${emailsHTML}
        </div>
        ${directionHTML}
        ${trackTypesHTML}
        ${matchedBanner}
      </div>`;
  }).join('');
}
```
